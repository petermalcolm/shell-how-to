# du
# The -hr sorts human-readable sizes in reverse (largest first).

# very specific directories:
du -sh tests/ContactInfo tests/Content tests/Controllers  | sort -hr | less

# the contents of those directories
du -sh tests/ContactInfo/* tests/Content/* tests/Controllers/*  | sort -hr | less

# for subdirectories given a desired depth
du -h tests/**/**/** | sort -hr | less

# show EVERYTHING
du -ah tests/ | sort -hr | less

# show name and size of all C files in a specific directory.  Also display a grand total at the end:
du -ch /usr/src/sys/kern/*.c


# Flags for `du`
#      -a      Display an entry for each file in a file hierarchy.
#      -c      Display a grand total.
#      -h      “Human-readable” output.  Use unit suffixes: Byte, Kilobyte, Megabyte, Gigabyte, Terabyte and Petabyte based on powers of 1024.
#      -s      Display an entry for each specified file.  (Equivalent to -d 0)
